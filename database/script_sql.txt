-- =====================================================
-- Skull Brew - Catálogo de cervecería (MySQL 8+)
-- Solo catálogo: sin clientes, pedidos ni pagos.
-- Incluye usuarios del panel con roles.
-- =====================================================

CREATE DATABASE IF NOT EXISTS cerveceria_skull_brew
  CHARACTER SET utf8mb4
  COLLATE utf8mb4_unicode_ci;
USE cerveceria_skull_brew;

-- -----------------------------------------------------
-- Roles y usuarios del panel de administración
-- -----------------------------------------------------
CREATE TABLE roles (
  id          INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  nombre      VARCHAR(50) NOT NULL UNIQUE,    -- 'admin', 'encargado', 'vendedor'
  descripcion VARCHAR(255) NULL
) ENGINE=InnoDB;

CREATE TABLE usuarios (
  id            INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  rol_id        INT UNSIGNED NOT NULL,
  nombre        VARCHAR(100) NOT NULL,
  email         VARCHAR(150) NOT NULL UNIQUE,
  password_hash VARCHAR(255) NOT NULL,        -- generado con password_hash() de PHP
  activo        TINYINT(1) NOT NULL DEFAULT 1,
  ultimo_acceso TIMESTAMP NULL,
  created_at    TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at    TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  CONSTRAINT fk_usuarios_rol
    FOREIGN KEY (rol_id) REFERENCES roles(id),
  INDEX idx_usuarios_rol (rol_id)
) ENGINE=InnoDB;

-- -----------------------------------------------------
-- Categorías (jerárquicas: Sin alcohol > Gaseosas)
-- -----------------------------------------------------
CREATE TABLE categorias (
  id          INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  parent_id   INT UNSIGNED NULL,
  nombre      VARCHAR(100) NOT NULL,
  slug        VARCHAR(120) NOT NULL UNIQUE,
  descripcion TEXT NULL,
  orden       SMALLINT UNSIGNED NOT NULL DEFAULT 0,
  activa      TINYINT(1) NOT NULL DEFAULT 1,
  created_at  TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at  TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  CONSTRAINT fk_categorias_parent
    FOREIGN KEY (parent_id) REFERENCES categorias(id) ON DELETE SET NULL,
  INDEX idx_categorias_parent (parent_id)
) ENGINE=InnoDB;

-- -----------------------------------------------------
-- Estilos de cerveza (tabla de referencia)
-- -----------------------------------------------------
CREATE TABLE estilos_cerveza (
  id          INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  nombre      VARCHAR(80) NOT NULL UNIQUE,
  descripcion TEXT NULL
) ENGINE=InnoDB;

-- -----------------------------------------------------
-- Productos (datos comunes a cualquier bebida)
-- -----------------------------------------------------
CREATE TABLE productos (
  id            INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  categoria_id  INT UNSIGNED NOT NULL,
  nombre        VARCHAR(150) NOT NULL,
  slug          VARCHAR(170) NOT NULL UNIQUE,
  descripcion   TEXT NULL,
  es_alcoholico TINYINT(1) NOT NULL DEFAULT 0,
  destacado     TINYINT(1) NOT NULL DEFAULT 0,
  badge         VARCHAR(30) NULL,
  disponible    TINYINT(1) NOT NULL DEFAULT 1,  -- visible pero sin disponibilidad (ej. agotada)
  activo        TINYINT(1) NOT NULL DEFAULT 1,  -- borrado lógico / oculto en la web
  created_at    TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at    TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  CONSTRAINT fk_productos_categoria
    FOREIGN KEY (categoria_id) REFERENCES categorias(id),
  INDEX idx_productos_categoria (categoria_id),
  INDEX idx_productos_activo (activo, disponible),
  INDEX idx_productos_destacado (destacado)
) ENGINE=InnoDB;

-- -----------------------------------------------------
-- Detalle específico de cervezas (relación 1:1 con productos)
-- -----------------------------------------------------
CREATE TABLE cervezas_detalle (
  producto_id   INT UNSIGNED PRIMARY KEY,
  estilo_id     INT UNSIGNED NULL,
  abv           DECIMAL(4,2) NULL,          -- % de alcohol, ej. 5.50
  ibu           SMALLINT UNSIGNED NULL,     -- amargor
  srm           DECIMAL(4,1) NULL,          -- color
  maltas        VARCHAR(255) NULL,
  lupulos       VARCHAR(255) NULL,
  notas_cata    TEXT NULL,
  maridaje      VARCHAR(255) NULL,
  CONSTRAINT fk_cervezas_producto
    FOREIGN KEY (producto_id) REFERENCES productos(id) ON DELETE CASCADE,
  CONSTRAINT fk_cervezas_estilo
    FOREIGN KEY (estilo_id) REFERENCES estilos_cerveza(id) ON DELETE SET NULL,
  INDEX idx_cervezas_estilo (estilo_id)
) ENGINE=InnoDB;

-- -----------------------------------------------------
-- Presentaciones (formato + precio): lata, botella, growler, pinta, barril...
-- -----------------------------------------------------
CREATE TABLE presentaciones (
  id           INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  producto_id  INT UNSIGNED NOT NULL,
  formato      VARCHAR(50) NOT NULL,        -- 'Lata', 'Botella', 'Growler', 'Pinta'
  volumen_ml   INT UNSIGNED NULL,
  precio       DECIMAL(10,2) NOT NULL,
  sku          VARCHAR(50) NULL UNIQUE,
  disponible   TINYINT(1) NOT NULL DEFAULT 1,
  orden        SMALLINT UNSIGNED NOT NULL DEFAULT 0,
  created_at   TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at   TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  CONSTRAINT fk_presentaciones_producto
    FOREIGN KEY (producto_id) REFERENCES productos(id) ON DELETE CASCADE,
  INDEX idx_presentaciones_producto (producto_id)
) ENGINE=InnoDB;

-- -----------------------------------------------------
-- Imágenes de producto
-- -----------------------------------------------------
CREATE TABLE producto_imagenes (
  id           INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  producto_id  INT UNSIGNED NOT NULL,
  url          VARCHAR(255) NOT NULL,
  alt_text     VARCHAR(150) NULL,
  es_principal TINYINT(1) NOT NULL DEFAULT 0,
  orden        SMALLINT UNSIGNED NOT NULL DEFAULT 0,
  CONSTRAINT fk_imagenes_producto
    FOREIGN KEY (producto_id) REFERENCES productos(id) ON DELETE CASCADE,
  INDEX idx_imagenes_producto (producto_id)
) ENGINE=InnoDB;
